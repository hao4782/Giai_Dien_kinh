Mã Google Apps Script chuẩn

Mã Apps Script được cấu hình tự động xử lý lưu/đọc qua Web App URL.

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  function writeSheet(name, rows) {
    var sh = ss.getSheetByName(name) || ss.insertSheet(name);
    sh.clearContents();
    if (rows && rows.length > 0) {
      sh.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
    }
  }
  if (payload.sheets) {
    for (var sheetName in payload.sheets) {
      writeSheet(sheetName, payload.sheets[sheetName]);
    }
  }
  var meta = ss.getSheetByName('Data_System') || ss.insertSheet('Data_System');
  meta.clearContents();
  meta.getRange(1, 1).setValue(JSON.stringify(payload.rawState));
  return ContentService.createTextOutput(JSON.stringify({status: 'ok'})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var meta = ss.getSheetByName('Data_System');
  var raw = meta ? meta.getRange(1, 1).getValue() : '{}';
  return ContentService.createTextOutput(raw).setMimeType(ContentService.MimeType.JSON);
}
